#include "alu.hpp"


//  Допоміжна функція для встановлення флагів 
//  для побітових логічних операцій (AND, OR, XOR)
static inline void set_bit_operation_flags(Flags& flags, Byte result) {
    flags.c = false;                //  ніколи немає переповнення
    flags.z = (result == 0);        //  нульовий результат?
    flags.n = (result & 0x80) != 0; //  старший біт - одиниця?
}


//  Побітове додавання двох байтів
//  carry - перенесення 8го байта (чи виникло)
Byte ADD(Byte a, Byte b, Flags &flags){
    Byte result = 0;
    bool carry = false;          // вхідний перенос, локальний

    for (int i = 0; i < 8; ++i) {
        bool ab = (a >> i) & 1;
        bool bb = (b >> i) & 1;
        bool t  = ab ^ bb ^ carry;

        carry = (ab && bb) || (ab && carry) || (bb && carry);

        if (t)
            result = (Byte)(result | (1u << i));
    }

    flags.c = carry;                 // записуємо локальний перенос у флаг
    flags.z = (result == 0);
    flags.n = (result & 0x80) != 0;  // 0x80 = 128(10) = 1000_0000(2) - маска старшого біта

    return result;
}

//  Побітове віднімання двох байтів
//  borrow - запозичення з 9го байта (чи виникло)
Byte SUB(Byte a, Byte b, Flags &flags){
    Byte result = 0;
    bool borrow = false;         // вхідне запозичення, локальне

    for (int i = 0; i < 8; ++i) {
        bool ab = (a >> i) & 1;
        bool bb = (b >> i) & 1;
        bool t  = ab ^ bb ^ borrow;

        // borrow виникає, коли треба відняти, а біта не вистачає (відняти від 0 одиницю)
        borrow = (!ab && bb) || (!ab && borrow) || (bb && borrow);

        if (t)
            result = (Byte)(result | (1u << i));
    }

    flags.c = borrow;                // записуємо вихідне запозичення у флаг
    flags.z = (result == 0);
    flags.n = (result & 0x80) != 0;

    return result;
}

//  -----------------------------------
//  --- Побітові логічні операції   ---

Byte AND(Byte a, Byte b, Flags &flags){
    Byte result = (Byte)(a & b);
    set_bit_operation_flags(flags, result);

    /*  Example
        0010 1010
        0111 0110
        ==========
        0010 0010
    */

    return result;
}
Byte OR(Byte a, Byte b, Flags &flags){
    Byte result = (Byte)(a | b);
    set_bit_operation_flags(flags, result);

    /* Example
        0110 0100
        1010 0111
        ===========
        1110 0111
    */

    return result;
}
Byte XOR(Byte a, Byte b, Flags &flags){
    Byte result = (Byte)(a ^ b);
    set_bit_operation_flags(flags, result);

    /* Example
        0110 0100
        1010 0111
        ===========
        1100 0011
    */

    return result;
}
Byte NOT(Byte a, Flags &flags){
    Byte result = ~a;
    set_bit_operation_flags(flags, result);
    
    /* Example
        0100 1101
        ===========
        1011 0010
    */

    return result;
}
//  -------------------------------------------------------------



//  -----------------------
//  --- Зсуви (логічні) ---

Byte SHL(Byte a, Flags &flags){
    Byte result = (Byte)(a << 1);

    /* Example
        0100 1101 <<
        =============
        1001 1010
    */

    flags.c = (a & 0x80) != 0;       //  біт 7 виштовхнувся назовні
    flags.z = (result == 0);
    flags.n = (result & 0x80) != 0;

    return result;
}
Byte SHR(Byte a, Flags &flags){
    Byte result = (Byte)((a >> 1) | (a & 0x80));
    
    /* Example
        0100 1101 >>
        =============
        0010 0110
    */

    flags.c = (a & 0x01) != 0;
    flags.z = (result == 0);
    flags.n = (result & 0x80) != 0;
    
    return result;
}
//  ------------------------------------------------------------------



//  -------------------------------
//  --- Інкремент / декремент   ---

Byte INC(Byte a, Flags &flags){
    bool saved_c = flags.c;
    
    Byte result = ADD(a, 1, flags);
    flags.c = saved_c;               //  INC не впливає на C
    
    return result;
}
Byte DEC(Byte a, Flags &flags){
    bool saved_c = flags.c;
    Byte result = SUB(a, 1, flags);
    flags.c = saved_c;               //  DEC не впливає на C
    return result;
}

void dump_regs(const CPU& cpu) {
    cout << "PC = " << "0x" << hex << cpu.pc << dec << "\n";
    cout << "H = " << "0x" << hex << setw(2) << setfill('0') << cpu.H << dec << "  "
         << to_bin(cpu.H, 16, 4) << "\n";
    cout << "A = ";
    show_byte(cpu.a);
    cout << "B = ";
    show_byte(cpu.b);
    cout << "Z: " << cpu.f.z << "\n" << "N: " << cpu.f.n << "\n" << "C: " << cpu.f.c << "\n";
}

inline const char* bit(bool b) { return b ? "1" : "0"; }
string flags_state(const CPU& cpu) {
    return "   |   Z: " + string(bit(cpu.f.z)) + "  " + "N: " + string(bit(cpu.f.n)) + "  " +
           "C: " + string(bit(cpu.f.c));
}



static bool is_mem_end(CPU &cpu) {
    size_t idx = static_cast<size_t>(cpu.pc);
    if(idx >= MEM_SIZE){
        cout << "cpu.pc >= " << MEM_SIZE << ", value was " << cpu.pc << ". CPU was halted.\n";
        cpu.halted = true;
        return true;
    }
    return false;
}

// універсальний обробник невідомого опкоду для всіх default:
static void unknown_opcode(Byte opcode, CPU &cpu){
    cerr << "Unknown opcode 0x" << hex << static_cast<int>(opcode)
        << " at PC=0x" << cpu.pc << dec << "\n";
    cpu.halted = true;
}

//  загальна ф-я для обробки стрибків за умови
static void JMPX(bool flag, CPU &cpu){
    if(!flag){   //  якщо флаг НЕ виставлений - пропустити цю команду
        cpu.pc +=3;
        return;
    }
    //  інакше, флаг ВИСТАВЛЕНИЙ
    cpu.pc = get16(*cpu.mem, cpu.pc+1);
}





void step(CPU& cpu) {
    if (cpu.halted) {
        cout << "halted\n";
        return;
    }
    //  -- cpu.mem - це сам вказівник
    //  ~ cpu.mem = ...                                         | переміщення
    //  ~ cpu.mem++ / cpu.mem-- / cpu.mem += ... / cpu.mem -=   | переміщення через арифметику
    //  ~ *cpu.mem                                              | взяти значення
    Byte op = mem_get(*cpu.mem, cpu.pc);

    switch ((Op)op) {
        //  ------------------------------------------------------------------------------------
        // -- IO
        case Op::HALT:
            cpu.halted = true;
            cpu.pc++; // розмір HALT з таблиці — 1
            break;
        case Op::NOP:
            cpu.pc++;
            break;
        case Op::OUT:
            cout << to_character(cpu.a) << "\n";
            cpu.pc++;
            break;
        case Op::OUTN:
            cout << static_cast<int>(cpu.a) << " ";
            cpu.pc++;
            break;
        case Op::OUTS: {
            cpu.pc++;
            U16 addr = get16(*cpu.mem, cpu.pc);
            cpu.pc+=2;

            for(int i = 0; i < 256; i++){
                if(cpu.mem->data[addr + i] == '\0')
                    break;
                
                cout << static_cast<char>(cpu.mem->data[addr+i]);
            }

            break;
        }

        //  ------------------------------------------------------------------------------------
        // -- ALU
        case Op::ADD: {
            cout << "\n"
                 << "ADD(" << static_cast<int>(cpu.a) << ", " << static_cast<int>(cpu.b)
                 << ") = " << static_cast<int>(ADD(cpu.a, cpu.b, cpu.f)) << flags_state(cpu)
                 << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::SUB: {
            cout << "\n"
                 << "SUB(" << static_cast<int>(cpu.a) << ", " << static_cast<int>(cpu.b)
                 << ") = " << static_cast<int>(SUB(cpu.a, cpu.b, cpu.f)) << flags_state(cpu)
                 << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::AND: {
            cout << "\n"
                 << "AND = " << to_bin(AND(cpu.a, cpu.b, cpu.f)) << flags_state(cpu) << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::OR: {
            cout << "\n"
                 << "OR = " << to_bin(OR(cpu.a, cpu.b, cpu.f)) << flags_state(cpu) << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::XOR: {
            cout << "\n"
                 << "XOR = " << to_bin(XOR(cpu.a, cpu.b, cpu.f)) << flags_state(cpu) << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::NOT: {
            cout << "\n"
                 << "NOT(a) = " << to_bin(NOT(cpu.a, cpu.f)) << flags_state(cpu) << "\n"
                 << "NOT(b) = " << to_bin(NOT(cpu.b, cpu.f)) << flags_state(cpu) << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::SHL: {
            cout << "\n"
                 << "SHL(a) = " << to_bin(SHL(cpu.a, cpu.f)) << flags_state(cpu) << "\n"
                 << "SHL(b) = " << to_bin(SHL(cpu.b, cpu.f)) << flags_state(cpu) << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::SHR: {
            cout << "\n"
                 << "SHR(a) = " << to_bin(SHR(cpu.a, cpu.f)) << flags_state(cpu) << "\n"
                 << "SHR(b) = " << to_bin(SHR(cpu.b, cpu.f)) << flags_state(cpu) << "\n\n";
            cpu.pc++;
            break;
        }

        case Op::INC: {
            cpu.a = INC(cpu.a, cpu.f);
            cpu.pc++;
            break;
        }

        case Op::DEC: {
            cpu.a = DEC(cpu.a, cpu.f);
            cpu.pc++;
            break;
        }

        case Op::CMP: {
            (void)SUB(cpu.a, cpu.b, cpu.f);
            cpu.pc++;
            break;
        }

        //  ------------------------------------------------------------------------------------
        // -- DATA
        case Op::LOADI_A: {
            cpu.pc++;                           //  наступна комірка = значення, imm8
            if(is_mem_end(cpu)) return;
            cpu.a = mem_get(*cpu.mem, cpu.pc);  //  взяти значення з пам'яті
            cpu.pc++;                           //  наступна інструкція
            break;
        }
        case Op::LOADI_B: {
            cpu.pc++;                           //  наступна комірка = значення, imm8
            if(is_mem_end(cpu)) return;
            cpu.b = mem_get(*cpu.mem, cpu.pc);  //  взяти значення з пам'яті
            cpu.pc++;                           //  наступна інструкція
            break;
        }
        case Op::LOAD_A: {
            cpu.pc++;
            if(is_mem_end(cpu)) return;
            U16 addr = get16(*cpu.mem, cpu.pc);
            cpu.a = mem_get(*cpu.mem, addr);        //  у регістр А
            cpu.pc+=2;
            break;
        }
        case Op::LOAD_B: {
            cpu.pc++;
            if(is_mem_end(cpu)) return;
            U16 addr = get16(*cpu.mem, cpu.pc);
            cpu.b = (Byte)mem_get(*cpu.mem, addr);  //  у регістр B
            cpu.pc+=2;
            break;
        }
        case Op::STORE_A: {
            cpu.pc++;
            if(is_mem_end(cpu)) return;
            U16 addr = get16(*cpu.mem, cpu.pc);
            mem_set(*cpu.mem, addr, cpu.a);
            cpu.pc+=2;
            break;
        }
        case Op::STORE_B: {
            cpu.pc++;
            if(is_mem_end(cpu)) return;
            U16 addr = get16(*cpu.mem, cpu.pc);
            mem_set(*cpu.mem, addr, cpu.b);
            cpu.pc+=2;
            break;
        }
        case Op::MOV_AB: {
            cpu.a = cpu.b;
            cpu.pc++;
            break;
        }
        case Op::MOV_BA: {
            cpu.b = cpu.a;
            cpu.pc++;
            break;
        }
        case Op::LOADH: {
            cpu.pc++;
            if(is_mem_end(cpu)) return;
            U16 addr = get16(*cpu.mem, cpu.pc);
            cpu.H = addr;
            cpu.pc+=2;
            break;
        }
        case Op::LOAD_AH: {
            cpu.a = mem_get(*cpu.mem, cpu.H);
            cpu.pc++;
            break;
        }
        case Op::STORE_HA: {
            mem_set(*cpu.mem, cpu.H, cpu.a);
            cpu.pc++;
            break;
        }
        case Op::INC_H: {
            cpu.H++;
            cpu.pc++;
            break;
        }
        case Op::DEC_H: {
            cpu.H--;
            cpu.pc++;
            break;
        }
        case Op::H_LOW: {
            /* NOT THIS LAB */
            break;
        }

        //  ------------------------------------------------------------------------------------
        // -- JUMP
        case Op::JMP:
            cpu.pc = get16(*cpu.mem, cpu.pc+1);
            break;

        case Op::JZ:
            JMPX(cpu.f.z, cpu);
            break;

        case Op::JNZ:
            JMPX(!cpu.f.z, cpu);
            break;

        case Op::JC:
            JMPX(cpu.f.c, cpu);
            break;

        case Op::JNC:
            JMPX(!cpu.f.c, cpu);
            break;

        //  ------------------------------------------------------------------------------------
        // -- DISPLAY
        case Op::SHOW:
            display::show(*cpu.mem);
            cpu.pc++;
            break;

        case Op::PLOT:
            display::plot(*cpu.mem, cpu.a, cpu.b);
            cpu.pc++;
            break;

        case Op::UNPLOT:
            display::unplot(*cpu.mem, cpu.a, cpu.b);
            cpu.pc++;
            break;

        case Op::CLS:
            display::clear(*cpu.mem);
            cpu.pc++;
            break;

        //  ------------------------------------------------------------------------------------
        // -- STACK
        case Op::PUSH_A:
            if (!push(cpu.stack, cpu.a)) { cerr << "Stack overflow at PC=...\n"; cpu.halted = true; return; }
            cpu.pc++;
            break;
        case Op::POP_A:
            if (!pop(cpu.stack, cpu.a)) { cerr << "Stack underflow ...\n"; cpu.halted = true; return; }
            cpu.pc++;
            break;

        //  ------------------------------------------------------------------------------------
        default:
            unknown_opcode(op,cpu);
            break;

    }
}

//#pragma once
#ifndef _CPU_HPP_
#define _CPU_HPP_

#include <iostream>
using namespace std;

#include "memory.hpp"
#include "stack.hpp"


enum class Op : Byte {
    // --- IO
    HALT      = 0x00,  //  Stop VM (exit)   0
    NOP       = 0x01,  //  No operation     1
    OUT       = 0x02,  //  Print A as symbol
    OUTN      = 0x03,  //  Print A as DEC number +space 
    OUTS      = 0x04, //  Print bytes from Addr while not '\0'

    // --- ALU
    ADD       = 0x10,  //  +    16
    SUB       = 0x11,  //  -    17
    AND       = 0x12,  //  &&   18
    OR        = 0x13,  //  ||   19
    XOR       = 0x14,  //  %2   20
    NOT       = 0x15,  //  !    21
    SHL       = 0x16,  //  <<   22
    SHR       = 0x17,  //  >>   23
    INC       = 0x18,  //  ++   24
    DEC       = 0x19,  //  --   25
    CMP       = 0x1a,  //  A-B, set Z/N/C

    // --- DATA
    LOADI_A   = 0x20, //  Load 8 bits to 
    LOADI_B   = 0x21, //  Load 8 bits to 
    LOAD_A    = 0x22, //  Load 8 bits to 
    LOAD_B    = 0x23, //  Load 8 bits to 
    STORE_A   = 0x24, //  Put A to mem[ad
    STORE_B   = 0x25, //  Put B to mem[ad
    MOV_AB    = 0x26, //  A = B
    MOV_BA    = 0x27, //  B = A
    LOADH     = 0x28, //  Load 16 bit to 
    LOAD_AH   = 0x29, //  A = mem[H]
    STORE_HA  = 0x2a, //  mem[H] = A
    INC_H     = 0x2b, //  H +1
    DEC_H     = 0x2c, //  H -1
    H_LOW     = 0x2d, //  low byte from H

    // --- JUMP
    JMP       = 0x30,  // jump anyway
    JZ        = 0x31,  // jump, if Z=1
    JNZ       = 0x32,  // jump, if Z=0
    JC        = 0x33,  // jump, if C=1
    JNC       = 0x34,  // jump, if C=0

    // --- DISPLAY
    CLS       = 0x40,  //  clear VRAM
    PLOT      = 0x41,  //  draw one pixel
    UNPLOT    = 0x42,  //  clear one pixel
    SHOW      = 0x43,  //  print VRAM in console

    // --- STACK
    PUSH_A    = 0x50,  // mem[SP] = A; SP = SP - 1
    POP_A     = 0x51,  // SP = SP + 1; A = mem[SP]
    PUSH_B    = 0x52,  // ...same
    POP_B     = 0x53,  // ...as for A
    CALL      = 0x54,  // покласти адресу повернення (2 байти), потім PC = addr
    RET       = 0x55,  // зняти 2 байти в PC
    PUSH_H    = 0x56,  // opt покласти H (2 байти: молодший, потім старший)
    POP_H     = 0x57,  // opt зняти 2 байти в H
};

namespace display {

    void show(const Memory& mem) {
        for (int y = 0; y < VRAM_ROWS; y++) {
            for (int x = 0; x < VRAM_COLS; x++) {
                Byte cell = mem.data[VRAM_LO + (y * 8) + (x / 8)];
                bool lit = (cell >> (7 - (x % 8))) & 1;
                cout << (lit ? VRAM_SYMBOL_FILL : VRAM_SYMBOL_EMPTY);
            }
            cout << '\n';
        }
    }

    static void plot_op(Memory& mem, int x, int y, bool set = true){
        if (x < 0 || x >= VRAM_COLS || y < 0 || y >= VRAM_ROWS)
            return;

        int address = VRAM_LO + (y * 8) + (x / 8);
        int bit = 7 - (x % 8);

        if(set)
            //  Для вставновлення "1" ==> операція АБО
            mem.data[address] |= ((Byte)1 << bit);
        else
            //  Для скидання ==> операція І-НЕ
            mem.data[address] &= ~((Byte)1 << bit);
    }

    void plot(Memory& mem, int x, int y) {
        plot_op(mem, x, y, true);
    }

    void unplot(Memory& mem, int x, int y) {
        plot_op(mem, x, y, false);
    }

    void clear(Memory& mem) {
        for (int i = 0; i < VRAM_ROWS*VRAM_COLS / 8; i++) {
            mem.data[VRAM_LO + i] = 0x00;
        }
    }

}

// How many bytes we print on one line. Try changing it to 8 and rebuilding.
const size_t BYTES_PER_LINE = 32;

// Is this byte something a terminal can print as a glyph?
// 0x20 is space, 0x7E is '~'. Everything outside that range we show as '.'.
static bool is_printable(Byte b) { return b >= 0x20 && b <= 0x7E; }

void dump(const Memory& mem) {
    // hex switches the stream to hexadecimal; setfill/setw pad with zeros
    // so every number is the same width and the columns line up.
    for (size_t row = 0; row < MEM_SIZE; row += BYTES_PER_LINE) {

        // The address column: 0000, 0010, 0020, ...
        cout << hex << setfill('0') << setw(4) << row << "  ";
        //  0, 16, 32, 48, ....

        // The hex column: 16 bytes, two digits each.
        for (size_t col = 0; col < BYTES_PER_LINE; ++col) {
            Byte b = mem.data[row + col];
            cout << setw(2) << static_cast<int>(b) << ' ';
        }

        cout << " |";

        // The ASCII gutter.
        for (size_t col = 0; col < BYTES_PER_LINE; ++col) {
            Byte b = mem.data[row + col];
            // TODO(lab-01, M2): when the byte IS printable, print the byte
            //                   itself instead of the dot. One token changes.
            //                   Hint: a Byte sent to cout prints as a
            //                   character already - that is the whole joke of
            //                   Lab 1. Right now every byte looks unprintable.
            if (is_printable(b)) {
                cout << b;
            } else {
                cout << '.';
            }
        }

        cout << "|\n";
    }

    // Put the stream back to decimal, or every number you print later is hex.
    cout << dec << setfill(' ');
}





string to_bin(long long n, int bits, int separate, bool show_prefix) {
    string res = show_prefix ? "0b" : "";
    for (int i = bits - 1; i >= 0; --i) {
        //  Transform data with binary operations
        res += (n & (1LL << i)) ? '1' : '0';

        if(i % separate == 0 && i > 0)
            res += "_";
    }
    return res;
}

string to_character(Byte b) {
    return is_printable(b) ? string(1, (char)b) : ".";
}

// Turn a word into a number. Accepts decimal (65) and hex (0x41).
// Returns false if the word is not a number at all.
static bool parse_number(const string& word, long& out) {
    try {
        size_t used = 0;
        // base 0 means: look at the prefix. "0x41" is hex, "65" is decimal.
        out = stol(word, &used, 0);
        return used == word.size(); // reject things like "12abc"
    } catch (...) {
        return false;
    }
}

void cmd_reg(CPU& cpu, istringstream& words) {
    string r, v;
    long value = 0;
    
    if (
        !(words >> r) || !(words >> v) ||   //  чи достатньо елементів у команді?
        !(r == "a" || r == "b") ||          //  r є "а" або "b"?
        !parse_number(v, value) ||          //  чи v є цілим числом?
        value < 0 || value > 255            //  чи v в межах 0-255
    ) {
        cout << "usage: reg <a|b> <0..255>\n";
        return;
    }
    
    cpu.a = (r == "a") ? (Byte)value : cpu.a;
    cpu.b = (r == "b") ? (Byte)value : cpu.b;
}

void cmd_run(CPU& cpu) {
    size_t steps = 0;

    if (cpu.halted) {
        cerr << "Error: CPU halted\n";
        return;
    }

    while (!cpu.halted) {

        if(steps >= MAX_STEPS){
            cout << "MAX_STEPS (" << MAX_STEPS << ") limit reached, run stopped\n";
            break;
        }

        step(cpu);
        steps++;
    }   
}

//  find <lo> <hi> <byte> <?deepth=8>
//  <lo> <hi> - це адреса, з якої почати пошук (глибиною 8 байт)
//  <byte> - яке значення шукаємо?
void cmd_find(Memory& mem, istringstream &words, bool first_only){
    string l, h, b, d;
    long low, high, search_byte;
    int deepth = 8;

    if( //  зчитування слів, перетворення у числа
        !(words >> l) || !(words >> h) || !(words >> b) ||
        !parse_number(l, low) || !parse_number(h, high) || !parse_number(b, search_byte)
    ) {
        cout << "usage: find <lo> <hi> <byte> <?deepth=8>\n";
        return;
    }

    long ldeepth = 0;       //  зчитування глибини пошуку (опціонально)
    if(words >> d) {
        parse_number(d, ldeepth);       //  перетворення у число

        if(ldeepth > 0 && (size_t)ldeepth < MEM_SIZE){
            //  якщо більше 0 та менше за розмір пам'яті - прийняти значення
            deepth = ldeepth;
        }
    }

    //  збірка 16-адреси з двох байтів (8)
    U16 addr = 0;
    addr = addr | low;
    addr = addr | (high << 8);

    //  перетворення шуканого байту у число
    long byte_value = static_cast<long>(search_byte);
    if(byte_value < -1 || byte_value > 255) {    //  перевірка меж
        cout << "byte must not be 0-255 of -1 (non-zero search), got " << byte_value << "\n";
        return;
    }

    bool finded = false;    //  чи знайдено хоча б один збіг?

    //  цикл пошуку: для addr та наступних deepth адрес
    for(int i = 0; i < deepth; i++){    //  до глибини пошуку
        if((size_t)(addr + i) >= MEM_SIZE)        //  чи кінець пам'яті?
            break;

        Byte mem_value = mem.data[addr + i];    //  взяти значення

        //  якщо співпало
        if(
            (static_cast<long>(mem_value) == search_byte) ||            //  збіг
            (search_byte == -1 && static_cast<long>(mem_value) != 0)    //  ненульове значення, якщо пошук за -1
        
        ) {
            cout << "Value 0x" << hex << static_cast<long>(mem_value) << " was finded at address 0x" << (addr+i) << dec << "\n";
            finded = true;

            if(first_only)
                return;
        }
    }

    //  якщо жодного збігу
    if(!finded) {
        cout << "Value 0x" << hex << search_byte << " was NOT finded at 0x" << addr << " - 0x" << addr+deepth << dec << "\n";
    }
}

void cmd_tick(Sprite (&sprites)[8], Memory& mem){

    //  Clear display buffer
    display::clear(mem);

    for(int i = 0; i < 8; i++){

        if(
            sprites[i].x >= display::VRAM_COLS || sprites[i].y >= display::VRAM_ROWS
        ){
            continue;
        }

        //  calculate memory address
        int address = display::VRAM_LO + (sprites[i].y * 8) + (sprites[i].x / 8);
        int bit = 7 - (sprites[i].x % 8);

        //  set sprite to memory
        mem.data[address] |= ((Byte)1 << bit);

        uint8_t nextX = sprites[i].x + sprites[i].vx;
        uint8_t nextY = sprites[i].y + sprites[i].vy;

        //  sprite moving
        if(nextX < display::VRAM_COLS) sprites[i].x = nextX;
        if(nextY < display::VRAM_ROWS) sprites[i].y = nextY;

        // check display edges collision
        if(
            (sprites[i].x >= display::VRAM_COLS-1 && sprites[i].vx > 0) ||
            (sprites[i].x <= 0 && sprites[i].vx < 0)
        )
            sprites[i].vx *= -1;

        if(
            (sprites[i].y >= display::VRAM_ROWS-1 && sprites[i].vy > 0) ||
            (sprites[i].y <= 0 && sprites[i].vy < 0)
        )
            sprites[i].vy *= -1;
        
    }

    //  Show display
    display::show(mem);
}
void cmd_tick_loop(Sprite (&sprites)[8], Memory& mem, istringstream& words){
    string c;
    long lc;
    if(!(words >> c) || !parse_number(c, lc)){
        cout << "usage: tick_loop <c>, where c > 0\n";
        return;
    }
    
    for(int i = 0; i < (int)lc; i++){
        system("clear");
        cmd_tick(sprites, mem);
        usleep(TICK_DELAY * 1000);
    }
}

using namespace std;

const int SIZE = 4095;

Byte mem_get(const Memory& mem, size_t addr) {
    
    if(addr > SIZE)
        return 0;

    return mem.data[addr];
}
U16 get16(const Memory& mem, size_t addr) {
    Byte low = mem_get(mem, addr);
    Byte high = mem_get(mem, addr + 1);
    return (U16)(low | (high << 8));
}


bool mem_set(Memory& mem, size_t addr, Byte value) {
    
    if(addr > SIZE)
        return false;

    mem.data[addr] = value;
    return true;
}
bool set16(Memory& mem, size_t addr, U16 value){

    if(addr >= SIZE)
        return false;

    Byte low = value;           //  автоматично відріже 8 молодших біт (усічення)
    Byte high = value >> 8;

    mem.data[addr] = low;
    mem.data[addr + 1] = high;

    return true;
}


bool inc(Memory& mem, size_t addr){

    if(addr > SIZE)
        return false;
    
    Byte b = mem_get(mem, addr);
    b++;
    return mem_set(mem, addr, b);
}

//src/stack.cpp
#include "stack.hpp"


void stack_init(Stack& s, Memory& mem) {
    s.mem = &mem;
    s.sp = STACK_HI;
}



bool push(Stack& s, Byte v) {
    if (s.sp < STACK_LO) return false;   // нижче 0xF00
    
    mem_set(*s.mem, s.sp, v);            // запис у пам'ять
    s.sp--;                              // зменшення адреси
    return true;
}

bool pop(Stack& s, Byte& out) {
    if (s.sp >= STACK_HI) return false;  // пустий стек

    s.sp++;                              // збільшення адреси
    out = mem_get(*s.mem, s.sp);         // читання з адреси
    return true;
}



U16 stack_sp(const Stack& s) {
    return s.sp;
}
